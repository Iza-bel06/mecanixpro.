document.getElementById('ordenForm').addEventListener('submit', async function(e) { e.preventDefault(); const servicios = []; document.querySelectorAll('#tabla-servicios tr').forEach(tr => { const desc = tr.querySelector('.serv-desc').value; const precio = tr.querySelector('.serv-precio').value; if (desc.trim() !== '') servicios.push({ descripcion: desc, precio: parseFloat(precio) || 0 }); }); const repuestos = []; document.querySelectorAll('#tabla-repuestos tr').forEach(tr => { const desc = tr.querySelector('.rep-desc').value; const precio = tr.querySelector('.rep-precio').value; if (desc.trim() !== '') repuestos.push({ descripcion: desc, precio: parseFloat(precio) || 0 }); }); const payload = { cliente: { tipo_doc: document.getElementById('tipo_doc').value, num_doc: document.getElementById('num_doc').value, nombre: document.getElementById('nombre_cliente').value, telefono: document.getElementById('telefono').value, correo: document.getElementById('correo').value, direccion: document.getElementById('direccion').value }, vehiculo: { placa: document.getElementById('placa').value, marca: document.getElementById('marca').value, modelo: document.getElementById('modelo').value, anio: document.getElementById('anio').value, km: document.getElementById('km').value, color: document.getElementById('color').value, chasis: document.getElementById('chasis').value, motor: document.getElementById('motor').value }, servicios: servicios, repuestos: repuestos }; try { const response = await fetch('/generar-orden', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (response.ok) { const htmlTicket = await response.text(); const nuevaVentana = window.open('', '_blank'); nuevaVentana.document.write(htmlTicket); nuevaVentana.document.close(); } else { alert('Hubo un error al generar el ticket.'); } } catch (error) { console.error('Error:', error); alert('Error de conexión al generar la orden.'); } });
